@Controller
@RequestMapping("/finance/settlements")
@RequiredArgsConstructor
public class FinanceSettlementWebController {

    private static final String BASE = "/finance/settlements";
    private static final String LIST_VIEW = "finance/settlements/list";
    private static final String DETAIL_VIEW = "finance/settlements/detail";

    private final SettlementService settlementService;

    @GetMapping
    public String list(
            @ModelAttribute("filter") SettlementFilter filter,
            @RequestParam(defaultValue = "0") int page,
            @RequestParam(required = false) Integer size,
            Model model) {
        int pageSize = CatalogPageSupport.normalizeSize(size);
        Page<SettlementResponse> result = CatalogPageSupport.lastPageIfBeyond(
                settlementService.search(filter, page, pageSize),
                last -> settlementService.search(filter, last, pageSize));
        model.addAttribute("page", result);
        model.addAttribute("counts", settlementService.countByStatus());
        model.addAttribute("size", pageSize);
        return LIST_VIEW;
    }

    @PostMapping("/{id:\\d+}/payout")
    public String payout(
            @PathVariable Long id,
            @Valid @ModelAttribute PayoutRequest form,
            BindingResult errors,
            RedirectAttributes flash) {
        if (errors.hasErrors()) {
            flash.addFlashAttribute("flashError", errors.getFieldError().getDefaultMessage());
            return "redirect:" + BASE + "/" + id;
        }
        try {
            settlementService.payout(id, form, currentUserId());
            flash.addFlashAttribute("flashMessage", "Đã chi trả chủ xe.");
        } catch (BusinessException ex) {
            flash.addFlashAttribute("flashError", ex.getMessage());
        }
        return "redirect:" + BASE + "/" + id;
    }
}

Quy tắc:
- Giữ /{id:\\d+} để URL sai định dạng ra 404, không ra 500.
- Không viết nghiệp vụ trong controller. Controller chỉ nhận form, gọi service, gắn flash.
- Không bind entity lên form, luôn dùng *Request.
- Sau POST luôn redirect (PRG). Flash: flashMessage, flashError.
- Người thao tác lấy từ SessionAuthUtils.getUserId(session), không nhận từ form.
- Model list giữ tên: page, counts, q, status, staff, dateFrom, dateTo, amountMin, amountMax,
  size, staffs, base, statuses. sales/deal-parts.html đang đọc các tên này.

public interface SettlementService {

    Page<SettlementResponse> search(SettlementFilter filter, int page, int size);

    Map<SettlementStatus, Long> countByStatus();

    SettlementResponse get(Long id);

    /** @throws BusinessException when the settlement is not a draft */
    void submit(Long id, Long actorUserId);

    /** @throws BusinessException when the settlement is not waiting for review */
    void decide(Long id, SettlementDecisionRequest form, Long actorUserId);

    /** @throws BusinessException when the settlement is not approved */
    void payout(Long id, PayoutRequest form, Long actorUserId);
}

@Service
@RequiredArgsConstructor
public class SettlementServiceImpl implements SettlementService {

    private final SettlementRepository settlementRepository;
    private final PayoutRepository payoutRepository;
    private final AuditService auditService;

    @Override
    @Transactional(readOnly = true)
    public SettlementResponse get(Long id) {
        return settlementRepository.findDetailById(id)
                .map(SettlementServiceImpl::toDto)
                .orElseThrow(() -> new ResourceNotFoundException("Quyết toán", id));
    }

    private Settlement load(Long id) {
        return settlementRepository.findDetailById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Quyết toán", id));
    }

    private static SettlementResponse toDto(Settlement s) {
        return SettlementResponse.builder().id(s.getId()).code(s.getCode()).build();
    }
}

Quy tắc:
- Đọc: @Transactional(readOnly = true). Ghi: @Transactional.
- Không thấy: ResourceNotFoundException("Tên tiếng Việt", id).
- Sai luật: BusinessException("... tiếng Việt ...", HttpStatus.CONFLICT).
- Mọi luật phải nằm ở Service. Giao diện ẩn nút chỉ là phụ, gửi thẳng POST vẫn phải bị chặn.
- Tiền luôn là BigDecimal, không dùng long, double.
- Thao tác đổi nhiều bảng phải trong cùng một @Transactional.
- Ghi audit: auditService.record(actor, "PAYOUT_SETTLEMENT", "Settlement", id, code).

public interface SettlementRepository extends JpaRepository<Settlement, Long> {

    @Query(value = """
            select s from Settlement s
              join fetch s.salesContract c
              join fetch c.reservation r
              join fetch r.vehicle v
            where (:status is null or s.status = :status)
              and (lower(s.code) like :pattern
                   or lower(c.code) like :pattern
                   or lower(v.vin) like :pattern)
            """,
            countQuery = """
            select count(s) from Settlement s
              join s.salesContract c
              join c.reservation r
              join r.vehicle v
            where (:status is null or s.status = :status)
              and (lower(s.code) like :pattern
                   or lower(c.code) like :pattern
                   or lower(v.vin) like :pattern)
            """)
    Page<Settlement> search(@Param("pattern") String pattern,
                            @Param("status") SettlementStatus status,
                            Pageable pageable);

    @Query("""
            select s from Settlement s
              join fetch s.salesContract c
              join fetch s.consignmentAgreement a
            where s.id = :id
            """)
    Optional<Settlement> findDetailById(@Param("id") Long id);

    @Query("select s.status, count(s) from Settlement s group by s.status")
    List<Object[]> countByStatus();
}

public interface PaymentRepository extends JpaRepository<Payment, Long> {

    @Query("select coalesce(sum(p.amount), 0) from Payment p where p.salesContract.id = :contractId")
    BigDecimal sumCollected(@Param("contractId") Long contractId);

    boolean existsByReferenceCodeIgnoreCase(String referenceCode);
}

- join fetch cho quan hệ hiện lên UI. countQuery KHÔNG fetch.
- Lọc nhiều điều kiện dùng Filter object và JPQL như SaleNewsServiceImpl.

6. LUẬT NGHIỆP VỤ PHẢI CODE TRONG SERVICE
--------------------------------------------------------------------------------
6.1 PaymentService
createRequest(contractId):
  - Hợp đồng phải SIGNED
  - Tạo PaymentRequest(amount = remainingAmount, purpose = "Thanh toán hợp đồng",
    status = PARTIAL), đổi hợp đồng thành ACTIVE

recordReceipt(contractId, form, actor):
  remaining = salePrice − paymentRepository.sumCollected(contractId)
  - amount <= remaining, sai: "Số tiền thu không được lớn hơn số còn lại."
  - existsByReferenceCodeIgnoreCase(reference), trùng: "Mã tham chiếu X đã được dùng."
  - Lưu Payment(amount, method, referenceCode, payerName = buyer, paidAt, status = PAID)
  - Nếu sumCollected == salePrice thì PaymentRequest.status = PAID
  - Thu đủ KHÔNG tự đổi Handover, Handover đọc từ tổng tiền

Tiền cọc là khoản thu đầu tiên: tạo Payment lúc tạo Reservation, xem mục 8.

6.2 RefundService
Điều kiện tạo: Reservation CANCELLED hoặc EXPIRED và quyết định là Hoàn cọc.
  - Refund(payment = Payment của cọc, reservation, amount = depositAmount)
  - status PENDING → PROCESSED, ghi processedAt
  - KHÔNG sửa Payment cũ
  - Giữ cọc thì không tạo Refund
  - Mỗi Reservation tối đa một Refund

6.3 SettlementService
openFor(contract), gọi bên trong closeSale:
  agreement = hợp đồng ký gửi ACTIVE của hồ sơ
  base = salePrice
  commission = base × commissionRate / 100, làm tròn 2 số lẻ HALF_UP
  fee = fixedFee, null thì 0
  net = base − commission − fee
  status = DRAFT
  approvedCostAmount, refundHeldAmount, depositHeldAmount để 0

BigDecimal commission = base.multiply(agreement.getCommissionRate())
        .divide(BigDecimal.valueOf(100), 2, RoundingMode.HALF_UP);

- Không có agreement ACTIVE: báo lỗi, KHÔNG dùng mặc định 1,5%
- Số tiền chỉ tính một lần, không có form sửa

submit: DRAFT → PENDING_REVIEW, xóa lý do trả lại
decide:
  - Chỉ PENDING_REVIEW
  - APPROVED: ghi người duyệt
  - REJECTED: bắt buộc lý do, status về DRAFT, lưu lý do vào Settlement.notes
  - Lưu lịch sử bằng Approval(type = SETTLEMENT, targetType = "Settlement", targetId)
payout:
  - Chỉ APPROVED, sai: "Chỉ chi trả khi quyết toán đã được duyệt."
  - Tạo Payout(amount = net, recipientName, bankName, accountNumberMasked,
    referenceCode, status = COMPLETED, paidAt)
  - Settlement PAID_OUT, Consignment COMPLETED
Manager không được xem DRAFT: search lọc status <> DRAFT, get trả 404.

6.4 HandoverService
Trạng thái tính khi đọc:
  paid = sumCollected >= salePrice
  transferred = TransferCase.status == COMPLETED
  CLOSED nếu hợp đồng CLOSED; COMPLETED nếu Handover đã xong;
  READY nếu paid && transferred; còn lại PENDING

confirm(contractId, form, actor):
  - Kiểm tra lại paid && transferred ngay trong service
    Sai: "Chỉ bàn giao khi đã thu đủ tiền và sang tên xong."
  - Tạo Handover và HandoverItem
  - Chữ ký: fileStorageService.uploadImage rồi lưu path, KHÔNG lưu base64 vào DB
  - status = COMPLETED, handedOverAt = now, receiptIssued = true

closeSale(contractId, actor), MỘT @Transactional:
  - Handover phải COMPLETED, kiểm tra lại đủ 3 mốc
  - SalesContract.status = CLOSED, saleClosed = true
  - Handover.status = CLOSED, saleCloseChecked = true
  - Consignment.status = SOLD
  - settlementService.openFor(contract)
